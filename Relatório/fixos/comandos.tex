% Adaptacao para ABNT NBR 10520:2023 (autor em caixa mista nas citacoes entre parenteses)
% Substitui ' e ' por ' and ' nas citacoes em texto ingles conforme recomendacao da banca
\makeatletter
\renewcommand{\abntrefinfo}[3]{%
  \if@filesw
    {%
     \let\protect\noexpand
     \StrSubstitute{#1}{ e }{ and }[\@authand]%
     \ifthenelse{\boolean{ABNThyperref}}{%
       \immediate\write\@auxout{\string\bibciteEXPL{\abntnextkey}{\@authand}}%
       \immediate\write\@auxout{\string\bibciteIMPL{\abntnextkey}{\@authand}}%
       \immediate\write\@auxout{\string\bibciteYEAR{\abntnextkey}{#3}}%
     }{%
       \immediate\write\@auxout{\string\bibcite{\abntnextkey++EXPL}{\@authand}}%
       \immediate\write\@auxout{\string\bibcite{\abntnextkey++IMPL}{\@authand}}%
       \immediate\write\@auxout{\string\bibcite{\abntnextkey++YEAR}{#3}}%
     }%
    }%
  \fi
}
\makeatother
