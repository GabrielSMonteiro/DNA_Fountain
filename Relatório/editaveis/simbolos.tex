\begin{simbolos}
  \item[$K$] Number of original source blocks or packets
  \item[$N$] Total number of encoded droplets
  \item[$N_{\text{cand}}$] Number of candidate oligonucleotides generated before screening
  \item[$N_{\text{tx}}$] Number of transmitted encoded oligonucleotides (passed screening)
  \item[$N_{\text{rx}}$] Number of encoded oligonucleotides received by the decoder
  \item[$L$] Length of source segments or blocks in bytes
  \item[$\epsilon$] Coding overhead ($\epsilon = (N - K)/K$)
  \item[$\epsilon_{\text{tx}}$] Nominal transmission overhead ($\epsilon_{\text{tx}} = (N_{\text{tx}} - K)/K$)
  \item[$\epsilon_{\text{rx}}$] Effective received overhead ($\epsilon_{\text{rx}} = (N_{\text{rx}} - K)/K$)
  \item[$d$] Droplet degree (number of combined source blocks)
  \item[{$\bar{d}_{\text{emp}}$}] Empirical sample mean degree of droplets
  \item[{$\mathbb{E}_\mu[D]$}] Theoretical expected mean degree under the Robust Soliton Distribution
  \item[$\rho(d)$] Probability mass function of the Ideal Soliton Distribution
  \item[$\tau(d)$] Corrective term of the Robust Soliton Distribution
  \item[$\mu(d)$] Normalized probability mass function of the Robust Soliton Distribution
  \item[$\beta$] Normalization factor of the RSD ($\beta = \sum_{d=1}^K [\rho(d) + \tau(d)]$)
  \item[$c$] Free tuning parameter of the RSD ($c > 0$)
  \item[$\delta$] Theoretical robustness parameter of the RSD (asymptotic bound on failure probability with $K + O(\sqrt{K}\ln^2(K/\delta))$ received packets)
  \item[$R$] Scale parameter associated with the expected ripple size ($R = c \ln(K/\delta)\sqrt{K}$)
  \item[$Q_{\max}$] Maximum observed length of the operational queue of processable droplets
  \item[$Q_{\min}$] Minimum length of the operational queue during the active decoding loop
  \item[$m$] Mode index of the corrective component $\tau(d)$ of the RSD ($m = \lfloor K/R \rfloor$)
  \item[$R_{\text{screen}}$] Rejection rate of the biological screening module
  \item[$H_{\max}$] Maximum allowable homopolymer run length
  \item[$p_{\text{loss}}$] Droplet erasure probability in the channel
  \item[$\oplus$] Bitwise Exclusive-OR operation (addition in $GF(2)$)
\end{simbolos}
