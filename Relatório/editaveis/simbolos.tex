\begin{simbolos}
  \item[$K$] Number of original input segments / packets
  \item[$N$] Number of generated encoded droplets
  \item[$\epsilon$] Coding overhead ($\epsilon = \frac{N - K}{K}$)
  \item[$d$] Droplet degree (number of combined segments)
  \item[$\rho(d)$] Ideal Soliton distribution probability density function
  \item[$\tau(d)$] Robust Soliton distribution correction term
  \item[$\mu(d)$] Robust Soliton distribution normalized probability density function
  \item[$c$] Robust Soliton distribution free tuning parameter ($c > 0$)
  \item[$\delta$] Upper bound on the decoding failure probability
  \item[$R$] Expected ripple size parameter in Robust Soliton distribution
  \item[$p_e$] Droplet erasure probability in the channel
  \item[$\oplus$] Bitwise XOR operation (addition in $GF(2)$)
\end{simbolos}
